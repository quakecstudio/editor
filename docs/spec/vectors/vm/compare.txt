# Comparisons and logical operators: vm.md VM-6 to VM-8. Format: README.md.

== eq_f
spec: VM-6.1, VM-6.2
globals: 100=2 101=2 102=3 103=-0
code:
  EQ_F 100 101 110
  EQ_F 100 102 111      # unequal: 0, no change
  EQ_F 103 104 112      # -0 == +0
  DONE
expect:
  1 EQ_F 110=1
  2 EQ_F
  3 EQ_F 112=1
  4 DONE
  done

== float_test_reads_bits_as_float
spec: VM-8.7
globals: 100=-0 101=i5
code:
  NOT_F 100 0 110       # -0 is false as a float
  IF 100 2              # but its bit pattern is non-zero, so IF jumps
  GOTO 2
  NOT_F 101 0 111       # an integer value read as a float is a subnormal, not zero
  DONE
expect:
  1 NOT_F 110=1
  2 IF
  4 NOT_F
  5 DONE
  done
