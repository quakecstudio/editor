== recursion_restores_locals
spec: VM-15.2, VM-16.3
globals: 100=2 120=1 121=i2
functions:
  2: first=4 parm_start=110 locals=1 parms=1
code:
  STORE_F 100 4
  CALL1 121
  DONE
  IFNOT 110 4           # function 2 (n): if n == 0 return
  SUB_F 110 120 4
  CALL1 121             # recurse with n - 1
  STORE_F 110 122       # n is restored after the call
  RETURN 110
expect:
  1 STORE_F 4=2
  2 CALL1 110=2
  4 IFNOT
  5 SUB_F 4=1
  6 CALL1 110=1
  4 IFNOT
  5 SUB_F 4=0
  6 CALL1 110=0
  4 IFNOT
  8 RETURN 110=1
  7 STORE_F 122=1
  8 RETURN 1=1 110=2
  7 STORE_F 122=2
  8 RETURN 1=2 110=0
  3 DONE 1=0
  done
