# Float and vector arithmetic: vm.md VM-4 and VM-5. Format: README.md.

== add_f
spec: VM-4.1
globals: 100=1.5 101=2.25
code:
  ADD_F 100 101 102
  DONE
expect:
  1 ADD_F 102=3.75
  2 DONE
  done

== add_f_rounds_to_nearest_even
spec: VM-4.1
globals: 100=16777216 101=1 103=16777216 104=3
code:
  ADD_F 100 101 102     # 2^24 + 1 is a tie: rounds to the even 2^24, no change
  ADD_F 103 104 105     # 2^24 + 3 is a tie: rounds to the even 2^24 + 4
  ADD_F 100 101 100     # c may be a: same result written back, no change
  DONE
expect:
  1 ADD_F 102=16777216
  2 ADD_F 105=16777220
  3 ADD_F
  4 DONE
  done

== sub_f
spec: VM-4.1
globals: 100=1 101=0.25 103=1
code:
  SUB_F 100 101 102

  DONE
expect:
  1 SUB_F 102=0.75
  2 SUB_F
  3 SUB_F 105=-1
  4 DONE
  done

== div_f_by_zero
spec: VM-4.2
globals: 100=1 101=0 102=-1 103=-0
code:
  DIV_F 100 101 110     # 1 / +0
  DIV_F 102 101 111     # -1 / +0
  DIV_F 100 103 112     # 1 / -0
  DIV_F 102 103 113     # -1 / -0
  DIV_F 101 100 114     # +0 / 1 is +0: no change
  DONE
expect:
  1 DIV_F 110=inf
  2 DIV_F 111=-inf
  3 DIV_F 112=-inf
  4 DIV_F 113=inf
  5 DIV_F
  6 DONE
  done
