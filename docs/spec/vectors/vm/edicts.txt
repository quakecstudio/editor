# The rows of edicts.md "Allocation and freeing" (EDICT-1.4, EDICT-1.7, EDICT-5.3, EDICT-6,
# EDICT-7), run on id's builtins (`builtins: engine`), with the host setting the clock between
# runs (`host:`). Format: README.md.
#
# Each vector starts from a new map: `edicts:` is `maxclients + 1` and nothing is free.
# Builtin numbers (BUILTIN-5): spawn 14, remove 15, nextent 47. PARM0 is slot 4, RETURN
# slot 1; `nextthink` is field 46, which ED_Free sets to -1 and ED_ClearEdict back to 0.

== alloc_after_the_clients
spec: EDICT-6.2, EDICT-1.4
builtins: engine
maxclients: 1
edicts: 2
globals: 120=i2 121=i3
functions:
  2: builtin=14
  3: builtin=47
host:
  clock 1
  run 1
code:
  CALL0 120             # spawn(): edict 2
  CALL0 120             # spawn(): edict 3, so num_edicts is 4
  STORE_ENT 1 4
  CALL1 121             # nextent(e3): edict 4 is past num_edicts, so the world
  DONE
expect:
  run 1
  1 CALL0 1=e2
  2 CALL0 1=e3
  3 STORE_ENT 4=e3
  4 CALL1 1=e0
  5 DONE
  done
