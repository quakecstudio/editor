# Stores, field loads, field addresses and pointer stores: vm.md VM-10 to VM-12.
# Format: README.md.

== store_f
spec: VM-10.1, VM-10.4
globals: 100=1.5
code:
  STORE_F 100 101 120   # c is ignored
  DONE
expect:
  1 STORE_F 101=1.5
  2 DONE
  done

== store_types_unchecked
spec: VM-10.1, VM-10.3
globals: 100=nan 101=-0 102=0x7f800001
code:
  STORE_S 100 110       # a float NaN copied by the string store, bit for bit
  STORE_FNC 101 111
  STORE_F 102 112       # a signalling NaN pattern is copied unchanged
  DONE
expect:
  1 STORE_S 110=0x7fc00000
  2 STORE_FNC 111=-0
  3 STORE_F 112=0x7f800001
  4 DONE
  done
